// register_file.v
// 64 registers x 32-bit width
// Two read ports (rs, rt) combinational read
// One write port (rd) synchronous on posedge clk when write_en is high.

module register_file (
    input  wire         clk,
    input  wire         write_en,
    input  wire [5:0]   rs_addr,
    input  wire [5:0]   rt_addr,
    input  wire [5:0]   rd_addr,
    input  wire [31:0]  data_in,
    output reg  [31:0]  rs_data,
    output reg  [31:0]  rt_data
);

    reg [31:0] regs [0:63];
    integer i;

    initial begin
        // initialize registers to known values (optional)
        for (i = 0; i < 64; i = i + 1) regs[i] = 32'h00000000;
    end

    // synchronous write
    always @(posedge clk) begin
        if (write_en) begin
            regs[rd_addr] <= data_in;
        end
    end

    // combinational read (present current contents)
    always @(*) begin
        rs_data = regs[rs_addr];
        rt_data = regs[rt_addr];
    end

endmodule
